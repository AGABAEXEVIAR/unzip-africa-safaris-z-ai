   462→                      <CarFeatureIcon name={feature.icon} />
   463→                    </div>
   464→                    <span className="font-label text-charcoal text-xs">{feature.title}</span>
   465→                  </div>
   466→                ))}
   467→              </div>
   468→            </Reveal>
   469→          </div>
   470→
   471→          {/* Right — Image Stack Swipe Effect */}
   472→          <Reveal variant="right" delay={0.2}>
   473→            <SafariCarImageStack images={safariCarImages} />
   474→          </Reveal>
   475→
   476→        </div>
   477→      </div>
   478→    </section>
   479→  );

   482→/* ===================== Image Stack Swipe Component ===================== */
   483→function SafariCarImageStack({ images }: { images: string[] }) {
   484→  const [stack, setStack] = useState<string[]>(images);
   485→  const [swipeDir, setSwipeDir] = useState<"left" | "right" | null>(null);
   486→  const [isAnimating, setIsAnimating] = useState(false);
   487→
   488→  // Swipe the top image to the back — it goes behind, next image shows
   489→  const swipeNext = (dir: "left" | "right") => {
   490→    if (isAnimating || stack.length <= 1) return;
   491→    setIsAnimating(true);
   492→    setSwipeDir(dir);
   493→    // After the animation completes, move the top image to the back
   494→    setTimeout(() => {
   495→      setStack((prev) => {
   496→        const next = [...prev];
   497→        const top = next.shift(); // Remove top
   498→        if (top) next.push(top); // Put it at the back
   499→        return next;
   500→      });
   501→      setSwipeDir(null);
   502→      setIsAnimating(false);
   503→    }, 500);
   504→  };
   505→
   506→  // Touch / mouse swipe handling
   507→  const touchStartX = useRef(0);
   508→  const containerRef = useRef<HTMLDivElement>(null);
   509→
   510→  const handleTouchStart = (e: React.TouchEvent) => {
   511→    touchStartX.current = e.touches[0].clientX;
   512→  };
   513→
   514→  const handleTouchEnd = (e: React.TouchEvent) => {
   515→    const deltaX = e.changedTouches[0].clientX - touchStartX.current;
   516→    if (Math.abs(deltaX) > 40) {
   517→      swipeNext(deltaX > 0 ? "right" : "left");
   518→    }
   519→  };
   520→
   521→  // Click to advance
   522→  const handleClick = () => {
   523→    swipeNext("left");
   524→  };
   525→
   526→  return (
   527→    <div
   528→      ref={containerRef}
   529→      className="relative w-full aspect-[4/5] max-w-[500px] mx-auto select-none"
   530→      onTouchStart={handleTouchStart}
   531→      onTouchEnd={handleTouchEnd}
   532→      onClick={handleClick}
   533→      style={{ cursor: "pointer" }}
   534→      data-cursor="view"
   535→    >
   536→      <style>{`
   537→        @keyframes stackSwipeLeft {
   538→          0% { transform: translateX(0) rotate(0deg) scale(1); opacity: 1; z-index: 10; }
   539→          100% { transform: translateX(-120%) rotate(-15deg) scale(0.9); opacity: 0; z-index: 0; }
   540→        }
   541→        @keyframes stackSwipeRight {
   542→          0% { transform: translateX(0) rotate(0deg) scale(1); opacity: 1; z-index: 10; }
   543→          100% { transform: translateX(120%) rotate(15deg) scale(0.9); opacity: 0; z-index: 0; }
   544→        }
   545→        .stack-img {
   546→          position: absolute;
   547→          inset: 0;
   548→          width: 100%;
   549→          height: 100%;
   550→          object-fit: cover;
   551→          transition: transform 0.4s cubic-bezier(0.16, 1, 0.3, 1);
   552→          will-change: transform, opacity;
   553→        }
   554→        .stack-img.swiping-left {
   555→          animation: stackSwipeLeft 0.5s cubic-bezier(0.16, 1, 0.3, 1) forwards;
   556→        }
   557→        .stack-img.swiping-right {
   558→          animation: stackSwipeRight 0.5s cubic-bezier(0.16, 1, 0.3, 1) forwards;
   559→        }
   560→      `}</style>
   561→
   562→      {/* Hint text */}
   563→      <div className="absolute -top-8 left-0 right-0 text-center">
   564→        <p className="font-eyebrow text-charcoal/40">Swipe or tap to view →</p>
   565→      </div>
   566→
   567→      {/* Stack of images — last in array is on top */}
   568→      {stack.map((src, idx) => {
   569→        const isTop = idx === 0;
   570→        const isSecond = idx === 1;
   571→        const isThird = idx === 2;
   572→
   573→        // Stack offset — cards behind the top one are slightly offset
   574→        const offset = isTop ? 0 : isSecond ? 12 : isThird ? 24 : 36;
   575→        const scale = isTop ? 1 : isSecond ? 0.95 : isThird ? 0.9 : 0.85;
   576→        const opacity = isTop ? 1 : isSecond ? 0.8 : isThird ? 0.6 : 0.4;
   577→
   578→        return (
   579→          <div
   580→            key={src}
   581→            className="absolute overflow-hidden bg-bone border border-border/30 shadow-lg"
   582→            style={{
   583→              top: `${offset}px`,
   584→              left: `${offset}px`,
   585→              right: `${-offset}px`,
   586→              bottom: `${-offset}px`,
   587→              zIndex: stack.length - idx,
   588→              transform: `scale(${scale})`,
   589→              opacity: opacity,
   590→              borderRadius: 0,
   591→            }}
   592→          >
   593→            <img
   594→              src={src}
   595→              alt={`Safari vehicle ${idx + 1}`}
   596→              className={`stack-img ${isTop && isAnimating ? (swipeDir === "right" ? "swiping-right" : "swiping-left") : ""}`}
   597→            />
   598→            {/* Gradient overlay */}
   599→            <div className="absolute inset-0 bg-gradient-to-t from-charcoal/40 via-transparent to-transparent" />
   600→            {/* Frame border */}
   601→            <div className="absolute inset-3 border border-cream/15 pointer-events-none" />
   602→          </div>
   603→        );
   604→      })}
   605→
   606→      {/* Counter */}
   607→      <div className="absolute -bottom-8 left-0 right-0 text-center">
   608→        <p className="font-label text-charcoal/40 text-xs">
   609→          {stack.length} photos — swipe to cycle
   610→        </p>
   611→      </div>
   612→    </div>
   613→  );
   614→}
