   446→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "activities" | "tripTypes" | "accommodationLevels" | "nationalParks">, value: string) => void;
   447→  clearFilters: () => void;

   450→  activitiesList: string[];
   451→  tripTypesList: string[];
   452→  accommodationLevelsList: string[];

   575→/* ===================== Mobile Filter Toggle ===================== */
   576→function MobileFilterToggle({
   577→  filters,
   578→  setFilters,
   579→  activeCount,
   580→  clearFilters,
   581→  toggleArrayFilter,
   582→  destinationsList,
   583→  activitiesList,
   584→  tripTypesList,
   585→  accommodationLevelsList,
   586→  nationalParksList,
   587→  priceMin,
   588→  priceMax,
   589→  durationMin,
   590→  durationMax,
   591→}: {
   592→  filters: FilterState;
   593→  setFilters: React.Dispatch<React.SetStateAction<FilterState>>;
   594→  activeCount: number;
   595→  clearFilters: () => void;
   596→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "activities" | "tripTypes" | "accommodationLevels" | "nationalParks">, value: string) => void;
   597→  destinationsList: string[];
   598→  activitiesList: string[];
   599→  tripTypesList: string[];
   600→  accommodationLevelsList: string[];
   601→  nationalParksList: string[];
   602→  priceMin: number;
   603→  priceMax: number;
   604→  durationMin: number;
   605→  durationMax: number;
   606→}) {
   607→  const [open, setOpen] = useState(false);
   608→
   609→  useEffect(() => {
   610→    if (open) {
   611→      document.body.style.overflow = "hidden";
   612→    } else {
   613→      document.body.style.overflow = "";
   614→    }
   615→  }, [open]);
   616→
   617→  return (
   618→    <div className="lg:hidden mb-8">
   619→      <div className="flex items-center justify-between gap-4">
   620→        <button onClick={() => setOpen(true)} className="btn-luxury flex-1 sm:flex-none">
   621→          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className="mr-2">
   622→            <path d="M22 3H2l8 9.46V19l4 2v-8.54L22 3z" />
   623→          </svg>
   624→          Filters {activeCount > 0 && `(${activeCount})`}
   625→        </button>
   626→        {activeCount > 0 && (
   627→          <button onClick={clearFilters} className="text-xs text-charcoal/60 hover:text-gold underline">Clear all</button>
   628→        )}
   629→      </div>
   630→
   631→      {open && (
   632→        <div className="fixed inset-0 z-[60] bg-charcoal/70 backdrop-blur-sm flex">
   633→          <div className="bg-canvas w-full max-w-md h-full overflow-y-auto p-6 ml-auto modal-scroll" style={{ overflowY: "auto", flex: "1 1 0%", minHeight: 0 }}>
   634→            <div className="flex items-center justify-between mb-6 pb-4 border-b border-border">
   635→              <p className="font-display text-2xl text-charcoal" style={{ fontFamily: "var(--font-cormorant), serif" }}>Filters</p>
   636→              <button onClick={() => setOpen(false)} aria-label="Close filters" className="text-charcoal/50 hover:text-charcoal">
   637→                <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5">
   638→                  <path d="M18 6 6 18M6 6l12 12" />
   639→                </svg>
   640→              </button>
   641→            </div>
   642→            <FilterSidebar
   643→              filters={filters}
   644→              setFilters={setFilters}
   645→              toggleArrayFilter={toggleArrayFilter}
   646→              clearFilters={clearFilters}
   647→              activeCount={activeCount}
   648→              destinationsList={destinationsList}
   649→              activitiesList={activitiesList}
   650→              tripTypesList={tripTypesList}
   651→              accommodationLevelsList={accommodationLevelsList}
   652→              nationalParksList={nationalParksList}
   653→              priceMin={priceMin}
   654→              priceMax={priceMax}
   655→              durationMin={durationMin}
   656→              durationMax={durationMax}
   657→            />
   658→            <button onClick={() => setOpen(false)} className="btn-luxury btn-luxury-gold w-full mt-6">Show Results</button>
   659→          </div>
   660→        </div>
   661→      )}
   662→    </div>
   663→  );
   664→}
