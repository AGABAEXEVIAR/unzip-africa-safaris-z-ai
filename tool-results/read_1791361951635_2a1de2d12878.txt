   353→  filters,
   354→  setFilters,
   355→  toggleArrayFilter,
   356→  clearFilters,

   367→}: {
   368→  filters: FilterState;
   369→  setFilters: React.Dispatch<React.SetStateAction<FilterState>>;
   370→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "activities" | "tripTypes" | "accommodationLevels" | "nationalParks">, value: string) => void;
   371→  clearFilters: () => void;

   374→  activitiesList: string[];
   375→  tripTypesList: string[];

   472→          <path d="M6 9l6 6 6-6" />
   473→        </svg>
   474→      </button>
   475→      {open && <div className="space-y-2">{children}</div>}
   476→    </div>

   485→        style={{ borderRadius: 0 }}
   486→      >
   487→        {checked && (
   488→          <svg width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="white" strokeWidth="3">
   489→            <path d="M20 6 9 17l-5-5" />
   490→          </svg>

   500→function MobileFilterToggle({
   501→  filters,
   502→  setFilters,
   503→  activeCount,
   504→  clearFilters,
   505→  toggleArrayFilter,
   506→  destinationsList,
   507→  activitiesList,

   510→  nationalParksList,
   511→  priceMin,
   512→  priceMax,
   513→  durationMin,
   514→  durationMax,

   516→  filters: FilterState;
   517→  setFilters: React.Dispatch<React.SetStateAction<FilterState>>;
   518→  activeCount: number;
   519→  clearFilters: () => void;
   520→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "activities" | "tripTypes" | "accommodationLevels" | "nationalParks">, value: string) => void;
