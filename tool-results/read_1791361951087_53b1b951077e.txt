   524→/* ===================== Mobile Filter Toggle ===================== */
   525→function MobileFilterToggle({
   526→  filters,
   527→  setFilters,
   528→  activeCount,
   529→  clearFilters,
   530→  toggleArrayFilter,
   531→  destinationsList,
   532→  tripTypesList,
   533→  priceMin,
   534→  priceMax,
   535→  durationMin,
   536→  durationMax,
   537→}: {
   538→  filters: FilterState;
   539→  setFilters: React.Dispatch<React.SetStateAction<FilterState>>;
   540→  activeCount: number;
   541→  clearFilters: () => void;
   542→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "tripTypes">, value: string) => void;

   552→  useEffect(() => {

   557→    }
   558→  }, [open]);
   559→
   560→  return (

   562→      <div className="flex items-center justify-between gap-4">

   571→            className="mr-2"

   573→            <path d="M22 3H2l8 9.46V19l4 2v-8.54L22 3z" />
   574→          </svg>
   575→          Filters {activeCount > 0 && `(${activeCount})`}
   576→        </button>
   577→        {activeCount > 0 && (
   578→          <button
   579→            onClick={clearFilters}
   580→            className="text-xs text-charcoal/60 hover:text-gold underline"
   581→          >
   582→            Clear all
   583→          </button>
   584→        )}
   585→      </div>

   589→          <div
   590→            className="bg-canvas w-full max-w-md h-full overflow-y-auto p-6 ml-auto modal-scroll"
   591→            style={{ overflowY: "auto", flex: "1 1 0%", minHeight: 0 }}
   592→          >
   593→            <div className="flex items-center justify-between mb-6 pb-4 border-b border-border">
   594→              <p
   595→                className="font-display text-2xl text-charcoal"
   596→                style={{ fontFamily: "var(--font-cormorant), serif" }}
   597→              >
   598→                Filters
   599→              </p>
   600→              <button
   601→                onClick={() => setOpen(false)}
   602→                aria-label="Close filters"
   603→                className="text-charcoal/50 hover:text-charcoal"
   604→              >
   605→                <svg
   606→                  width="20"
   607→                  height="20"
   608→                  viewBox="0 0 24 24"
   609→                  fill="none"
   610→                  stroke="currentColor"
   611→                  strokeWidth="1.5"
   612→                >
   613→                  <path d="M18 6 6 18M6 6l12 12" />
   614→                </svg>
   615→              </button>
   616→            </div>
   617→            <FilterSidebar
   618→              filters={filters}
   619→              setFilters={setFilters}
   620→              toggleArrayFilter={toggleArrayFilter}
   621→              clearFilters={clearFilters}
   622→              activeCount={activeCount}
   623→              destinationsList={destinationsList}
   624→              tripTypesList={tripTypesList}

   632→            </button>
