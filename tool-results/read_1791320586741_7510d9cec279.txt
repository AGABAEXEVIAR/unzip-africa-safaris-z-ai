    10→// Build filter option lists from the tour data
    11→const destinationsList = Array.from(new Set(tourPackages.map((t) => t.destination)));

    13→const tripTypesList = Array.from(new Set(tourPackages.map((t) => t.tripType)));
    14→const accommodationLevelsList = Array.from(new Set(tourPackages.map((t) => t.accommodationLevel)));

    23→  destinations: string[];
    24→  activities: string[];
    25→  tripTypes: string[];
    26→  accommodationLevels: string[];

    33→  destinations: [],
    34→  activities: [],
    35→  tripTypes: [],
    36→  accommodationLevels: [],
    37→  nationalParks: [],
    38→  priceMax: PRICE_MAX,
    39→  durationMax: DURATION_MAX,

    57→  const filteredTours = useMemo(() => {
    58→    let result = tourPackages.filter((t) => {
    59→      if (filters.destinations.length > 0 && !filters.destinations.includes(t.destination)) return false;
    60→      if (filters.activities.length > 0 && !filters.activities.some((a) => t.activities.includes(a))) return false;
    61→      if (filters.tripTypes.length > 0 && !filters.tripTypes.includes(t.tripType)) return false;
    62→      if (filters.accommodationLevels.length > 0 && !filters.accommodationLevels.includes(t.accommodationLevel)) return false;
    63→      if (filters.nationalParks.length > 0 && !filters.nationalParks.includes(t.nationalPark)) return false;
    64→      if (t.priceFrom > filters.priceMax) return false;
    65→      if (t.durationDays > filters.durationMax) return false;
    66→      return true;
    67→    });
    68→
    69→    if (sortBy === "price-low") result = [...result].sort((a, b) => a.priceFrom - b.priceFrom);
    70→    if (sortBy === "price-high") result = [...result].sort((a, b) => b.priceFrom - a.priceFrom);
    71→    if (sortBy === "duration") result = [...result].sort((a, b) => b.durationDays - a.durationDays);
    72→    if (sortBy === "recommended") result = [...result].sort((a, b) => (b.featured ? 1 : 0) - (a.featured ? 1 : 0));
    73→
    74→    return result;
    75→  }, [filters, sortBy]);

    92→    (filters.priceMax !== PRICE_MAX ? 1 : 0) +
    93→    (filters.durationMax !== DURATION_MAX ? 1 : 0);

    95→  return (
    96→    <div className="page-enter">
    97→      {/* ====================== HERO ====================== */}

   100→          <img

   102→            alt="Wildebeest migration crossing a river"

   104→          />

   107→
   108→        <div className="relative h-full flex flex-col items-center justify-center text-center px-6">
   109→          <motion.p

   111→            animate={{ opacity: 1, y: 0 }}
   112→            transition={{ duration: 1.2, delay: 0.4 }}
   113→            className="font-eyebrow text-gold-soft mb-8 tracking-[0.4em]"
   114→          >
   115→            Curated Journeys
   116→          </motion.p>
   117→          <motion.h1
   118→            initial={{ opacity: 0, y: 40 }}
   119→            animate={{ opacity: 1, y: 0 }}
   120→            transition={{ duration: 1.6, delay: 0.6, ease: [0.16, 1, 0.3, 1] }}
   121→            className="font-display text-cream text-[2.8rem] sm:text-[4rem] md:text-[5.5rem] lg:text-[6.5rem] leading-[0.95] tracking-tight max-w-[90%]"

   124→            All Tours

   131→          >

   290→/* ===================== Filter Sidebar ===================== */

   292→  filters,
   293→  setFilters,
   294→  toggleArrayFilter,
   295→  clearFilters,
   296→  activeCount,
   297→}: {
   298→  filters: FilterState;
   299→  setFilters: React.Dispatch<React.SetStateAction<FilterState>>;

   347→          </div>
   348→        </FilterSection>
   349→
   350→        {/* Duration */}

   478→/* ===================== Mobile Filter Toggle ===================== */
   479→function MobileFilterToggle({
   480→  filters,
   481→  setFilters,
   482→  activeCount,
   483→  clearFilters,
   484→  toggleArrayFilter,
   485→}: {
   486→  filters: FilterState;
   487→  setFilters: React.Dispatch<React.SetStateAction<FilterState>>;
   488→  activeCount: number;
   489→  clearFilters: () => void;
   490→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "activities" | "tripTypes" | "accommodationLevels" | "nationalParks">, value: string) => void;
   491→}) {
   492→  const [open, setOpen] = useState(false);
   493→
   494→  useEffect(() => {
   495→    if (open) {
   496→      document.body.style.overflow = "hidden";
   497→    } else {
   498→      document.body.style.overflow = "";
   499→    }
   500→  }, [open]);
   501→
   502→  return (
   503→    <div className="lg:hidden mb-8">
   504→      <div className="flex items-center justify-between gap-4">
   505→        <button
   506→          onClick={() => setOpen(true)}
   507→          className="btn-luxury flex-1 sm:flex-none"
   508→        >
   509→          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className="mr-2">
   510→            <path d="M22 3H2l8 9.46V19l4 2v-8.54L22 3z" />
   511→          </svg>
   512→          Filters {activeCount > 0 && `(${activeCount})`}
   513→        </button>
   514→        {activeCount > 0 && (
   515→          <button onClick={clearFilters} className="text-xs text-charcoal/60 hover:text-gold underline">
   516→            Clear all
   517→          </button>
   518→        )}
   519→      </div>
   520→
   521→      {open && (
