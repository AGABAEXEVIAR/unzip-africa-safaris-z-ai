   370→   370→  toggleArrayFilter: (key: keyof Pick<FilterState, "destinations" | "activities" | "tripTypes" | "accommodationLevels" | "nationalParks">, value: string) => void;
   371→   371→  clearFilters: () => void;

   373→   373→  destinationsList: string[];

   375→   375→  tripTypesList: string[];
   376→   376→  accommodationLevelsList: string[];
